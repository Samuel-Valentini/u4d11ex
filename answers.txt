SELECT * FROM clienti

2.
SELECT nome, cognome FROM clienti
WHERE anno_di_nascita=1982
ORDER BY nome ASC 

3.
SELECT iva, COUNT (*) FROM fatture WHERE iva=20 GROUP BY iva

oppure senza la colonna iva:

SELECT COUNT (*) FROM fatture WHERE iva=20 

4.
SELECT * FROM public.prodotti WHERE EXTRACT(YEAR FROM data_attivazione)=2017 AND (in_produzione=true OR in_commercio=true)

5.
SELECT * FROM fatture JOIN clienti ON fatture.id_cliente=clienti.numero_cliente WHERE importo<1000 

6.
SELECT numero_fattura, importo, iva, data_fattura, denominazione FROM fatture JOIN fornitori ON fatture.numero_fornitore=fornitori.numero_fornitore

7. visualizzando l'anno:
SELECT EXTRACT(YEAR FROM data_fattura), COUNT(*) FROM fatture WHERE iva=20 GROUP BY EXTRACT(YEAR FROM data_fattura)

senza vedere l'anno:
SELECT COUNT(*) FROM fatture WHERE iva=20 GROUP BY EXTRACT(YEAR FROM data_fattura)

8.
SELECT EXTRACT(YEAR FROM data_fattura), COUNT(*), SUM(importo) FROM fatture GROUP BY EXTRACT(YEAR FROM data_fattura)

9.
SELECT EXTRACT(YEAR FROM data_fattura) FROM fatture
WHERE tipologia='assistenza' GROUP BY EXTRACT(YEAR FROM data_fattura) HAVING COUNT(*)>2

10. 
SELECT regione, SUM(importo) FROM fatture 
JOIN clienti ON fatture.id_cliente=clienti.numero_cliente 
JOIN regioni ON clienti.regione_residenza=regioni.id_regione
GROUP BY regione

11.
SELECT COUNT(DISTINCT numero_cliente) FROM public.clienti 
JOIN fatture ON clienti.numero_cliente=fatture.id_cliente
WHERE anno_di_nascita=1980 AND importo>50

se voglio vedere l'anno:
SELECT anno_di_nascita, COUNT(DISTINCT numero_cliente) FROM public.clienti 
JOIN fatture ON clienti.numero_cliente=fatture.id_cliente
WHERE anno_di_nascita=1980 AND importo>50 GROUP BY anno_di_nascita
